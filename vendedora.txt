 let todosProdutosVen = []; // Armazena os produtos carregados para filtragem rápida

function carregarCatalogoVendedora() {
  const container = document.getElementById('ven-grade-produtos');
  const catBar = document.getElementById('ven-categorias-bar');
  container.innerHTML = '<p style="grid-column: span 2; text-align: center;">Carregando produtos...</p>';

  db.collection("produtos").get().then(snapshot => {
    todosProdutosVen = [];
    const categorias = new Set();

    if (snapshot.empty) {
      container.innerHTML = '<p style="grid-column: span 2; text-align: center;">Nenhum produto disponível.</p>';
      return;
    }

    snapshot.forEach(doc => {
      const p = doc.data();
      p.id = doc.id;
      todosProdutosVen.push(p);
      if (p.categoria) categorias.add(p.categoria);
    });

    // Renderizar Botões de Categoria
    catBar.innerHTML = `<button class="btn-cat active" onclick="filtrarCategoriaVen('todas', this)" style="padding: 6px 14px; border-radius: 20px; border: 1px solid var(--primary); background: var(--primary); color: white; font-size: 12px; white-space: nowrap; cursor: pointer;">Todas</button>`;
    categorias.forEach(cat => {
      catBar.innerHTML += `<button class="btn-cat" onclick="filtrarCategoriaVen('${cat}', this)" style="padding: 6px 14px; border-radius: 20px; border: 1px solid #ddd; background: #f5f5f5; color: #333; font-size: 12px; white-space: nowrap; cursor: pointer;">${cat}</button>`;
    });

    renderizarCardsVendedora(todosProdutosVen);
  }).catch(err => {
    console.error(err);
    container.innerHTML = '<p style="grid-column: span 2; text-align: center; color: red;">Erro ao carregar catálogo.</p>';
  });
}

function renderizarCardsVendedora(lista) {
  const container = document.getElementById('ven-grade-produtos');
  container.innerHTML = '';

  if (!lista.length) {
    container.innerHTML = '<p style="grid-column: span 2; text-align: center;">Nenhum produto nesta categoria.</p>';
    return;
  }

  lista.forEach(p => {
    let fotoUrl = p.foto || p.imagem || p.fotoUrl || p.img || '';

    if (p.variacoes && Array.isArray(p.variacoes)) {
      p.variacoes.forEach(v => {
        const qtdEstoque = Number(v.quantidade) || 0;
        
        if (qtdEstoque > 0) {
          const preco = Number(v.preco) || 0;
          const fotoFinal = v.foto || v.imagem || v.fotoUrl || fotoUrl || 'https://via.placeholder.com/150?text=Sem+Foto';

          container.innerHTML += `
            <div class="prod-list-card" style="padding: 10px; border-radius: 12px; border: 1px solid #eee; background: #fff; display: flex; flex-direction: column; justify-content: space-between;">
              <div style="width: 100%; height: 120px; border-radius: 8px; overflow: hidden; background: #f9f9f9; margin-bottom: 8px; cursor: pointer;" onclick="expandirFotoVen('${fotoFinal}')">
                <img src="${fotoFinal}" alt="${p.nome}" style="width: 100%; height: 100%; object-fit: cover;" onerror="this.src='https://via.placeholder.com/150?text=Sem+Foto';">
              </div>
              <b style="font-size: 13px; color: #333; line-height: 1.2; margin-bottom: 4px;">${p.nome}</b>
              <small style="color: #666; font-size: 11px;">Cor/Tam: <b>${v.cor || '-'} / ${v.tamanho || '-'}</b></small>
              <small style="color: ${qtdEstoque < 3 ? '#d9534f' : '#2e7d32'}; font-size: 11px; margin-bottom: 6px;">
                Estoque: <b>${qtdEstoque} un.</b>
              </small>
              <span style="color: var(--primary); font-weight: bold; font-size: 14px; margin-bottom: 8px;">R$ ${preco.toFixed(2)}</span>
              <button class="btn" style="padding: 8px; font-size: 12px; margin-top: auto;" onclick='adicionarAoCarrinhoVen("${p.id}", "${p.nome}", "${v.cor}", "${v.tamanho}", ${preco})'>
                <i class="fa-solid fa-cart-plus"></i> Adicionar
              </button>
            </div>
          `;
        }
      });
    }
  });
}

function filtrarCategoriaVen(categoria, btn) {
  // Destacar botão selecionado
  document.querySelectorAll('#ven-categorias-bar .btn-cat').forEach(b => {
    b.style.background = '#f5f5f5';
    b.style.color = '#333';
    b.style.border = '1px solid #ddd';
  });
  btn.style.background = 'var(--primary)';
  btn.style.color = 'white';
  btn.style.border = '1px solid var(--primary)';

  if (categoria === 'todas') {
    renderizarCardsVendedora(todosProdutosVen);
  } else {
    const filtrados = todosProdutosVen.filter(p => p.categoria === categoria);
    renderizarCardsVendedora(filtrados);
  }
}

 function adicionarAoCarrinhoVen(idProd, nome, cor, tamanho, preco) {
   venCarrinho.push({ idProd, nome, cor, tamanho, preco, quantidade: 1 });
   localStorage.setItem('ven_carrinho', JSON.stringify(venCarrinho));
   alert("Produto adicionado ao carrinho!");
 }

 function carregarCarrinhoVendedora() {
   const container = document.getElementById('ven-itens-carrinho');
   if (!venCarrinho.length) {
     container.innerHTML = '<p>Carrinho vazio.</p>';
     return;
   }
   container.innerHTML = '';
   venCarrinho.forEach((item, index) => {
     const preco = Number(item.preco) || 0;
     container.innerHTML += `
       <div class="prod-list-card" style="flex-direction:row; justify-content:space-between; align-items:center;">
         <div><b>${item.nome}</b> (${item.cor}/${item.tamanho})<br>R$ ${preco.toFixed(2)}</div>
         <button style="background:red; color:white; border:none; padding:5px 8px; border-radius:6px; cursor:pointer;" onclick="removerCarrinhoVen(${index})">X</button>
       </div>
     `;
   });
 }

 function removerCarrinhoVen(index) {
   venCarrinho.splice(index, 1);
   localStorage.setItem('ven_carrinho', JSON.stringify(venCarrinho));
   carregarCarrinhoVendedora();
 }

 function enviarPedidoParaLojaPrincipal() {
   if (!venCarrinho.length) {
     alert("Seu carrinho está vazio!");
     return;
   }
   const totalPedido = venCarrinho.reduce((acc, i) => acc + (Number(i.preco) || 0), 0);
   const comissao = totalPedido * 0.15;

   const pedidoData = {
     vendedora: venDados.nome,
     vendedoraNome: venDados.nome,
     telefone: venDados.telefone,
     pix: venDados.pix,
     itens: venCarrinho,
     total: totalPedido,
     comissao: comissao,
     data: new Date().toLocaleDateString('pt-BR'),
     dataCriacao: firebase.firestore.FieldValue.serverTimestamp(),
     status: 'Pendente na Matriz'
   };

   // CORRIGIDO: Envia apenas uma vez para a coleção de pedidos das vendedoras
   db.collection("pedidos").add(pedidoData)
     .then(() => {
       alert("Pedido enviado com sucesso para confirmação e separação na Toque Íntimo GS!");
       venCarrinho = [];
       localStorage.setItem('ven_carrinho', JSON.stringify(venCarrinho));
       mudarPainelVen('painel');
     })
     .catch(err => alert("Erro ao enviar: " + err));
 }

 function enviarCatalogoClienteWhatsapp() {

   const msg = encodeURIComponent(`Olá! Confira as novidades do catálogo Toque Íntimo GS enviado pela consultora ${venDados.nome}: ${urlCatalogo}`);
   window.open(`https://wa.me/?text=${msg}`, '_blank');
 }

 function carregarMetricasVendedora() {
   // CORRIGIDO: Busca na coleção "pedidos_vendedoras"
   db.collection("pedidos").where("telefone", "==", venDados.telefone).get().then(snapshot => {
     let qtdVendas = snapshot.size;
     let valVendas = 0;
     let valComissao = 0;
     snapshot.forEach(doc => {
       const d = doc.data();
       valVendas += Number(d.total) || 0;
       valComissao += Number(d.comissao) || 0;
     });
     document.getElementById('ven-qtd-vendas').innerText = qtdVendas;
     document.getElementById('ven-val-vendas').innerText = `R$ ${valVendas.toFixed(2)}`;
     document.getElementById('ven-val-comissao').innerText = `R$ ${valComissao.toFixed(2)}`;
   });
 }

  modal.style.display = 'flex';
}

           <span style="color:green; font-size:11px;">Status: ${d.status || 'Processando'}</span>
         </div>
       `;
     });
   });
